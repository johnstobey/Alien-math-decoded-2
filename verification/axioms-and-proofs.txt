    (∀ x ∈ S, ∀ y ∈ S, sourceOp x y ∈ S) → (∀ (b : B), b ∈ T ↔ ∃ a ∈ S, f a = b) → ∀ x ∈ T, ∀ y ∈ T, targetOp x y ∈ T :=
fun {A} {B} sourceOp targetOp f S T hpres hclosed himage x hx y hy =>
  Exists.casesOn ((himage x).mp hx) fun a h =>

            And.casesOn h fun hb right =>
              Eq.ndrec (motive := fun y => y ∈ T → targetOp (f a) y ∈ T)
                (fun hy =>
                  (himage (targetOp (f a) (f b))).mpr (Exists.intro (sourceOp a b) ⟨hclosed a ha b hb, hpres a b⟩))
                right hy)
        right hx
